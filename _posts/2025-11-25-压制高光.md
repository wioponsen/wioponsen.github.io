def suppress_highlight(img, mask, global_ratio=1, highlight_ratio=0.7):
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    h, s, v = cv2.split(hsv)

    lab = cv2.cvtColor(img, cv2.COLOR_BGR2Lab)
    l, a, b = cv2.split(lab)

    t_l, _ = cv2.threshold(l[mask>0], thresh=0,  maxval=255, type=cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    t_v, _ = cv2.threshold(v[mask>0], thresh=0, maxval=255, type=cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    initial_threshold = max(t_l, t_v)

    # mask_highlight = np.zeros_like(l)
    # mask_highlight[l>=initial_threshold] = 255
    # mask_lows = np.zeros_like(l)
    # mask_lows[s<50] = 255  # 高光区域 高v低s 特性
    # mask_highlight = mask_highlight & mask_lows  # 如果需要准确高光mask，可以使用这个

    # 按阈值调整
    l_pro = l.astype(np.float32)
    l_pro[l>=initial_threshold] = initial_threshold + (l_pro[l>=initial_threshold] - initial_threshold)*highlight_ratio
    l_pro = (l_pro.clip(0, 255)*global_ratio).astype(np.uint8)

    suppressed_lab = cv2.merge([l_pro, a, b])
    img_sup = cv2.cvtColor(suppressed_lab, cv2.COLOR_LAB2BGR)

    return img_sup
